CREATE TRIGGER [dbo].[trg_GenerarIdVideos]
ON [dbo].[VIDEOS]
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @UltimoId NVARCHAR(10);
    DECLARE @NuevoId NVARCHAR(10);
    DECLARE @Numero INT;

    -- Obtener el último ID en la tabla en el formato N00001, N00002, etc.
    SELECT TOP 1 @UltimoId = IdVideo
    FROM dbo.VIDEOS
    WHERE IdVideo LIKE 'V%' -- Filtra solo IDs que empiezan con 'N'
    ORDER BY IdVideo DESC;

    -- Extraer el número de la parte del ID actual (después de la letra "N")
    SET @Numero = CASE
                    WHEN @UltimoId IS NULL THEN 1
                    ELSE CAST(SUBSTRING(@UltimoId, 2, LEN(@UltimoId) - 1) AS INT) + 1
                  END;

    -- Generar el nuevo ID en el formato "N00001", "N00002", etc.
    SET @NuevoId = 'V' + RIGHT('0000' + CAST(@Numero AS NVARCHAR), 4);

    -- Actualizar el ID del nuevo registro en la tabla usando la tabla "inserted"
    UPDATE dbo.VIDEOS
    SET IdVideo = @NuevoId
    FROM dbo.VIDEOS v
    INNER JOIN inserted i ON v.IdVideo = i.IdVideo;
END;
GO

ALTER TABLE [dbo].[VIDEOS] ENABLE TRIGGER [trg_GenerarIdVideos]
GO
